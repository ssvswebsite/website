/* Left, middle and right thirds of the section. Icons take a band by turn
   rather than a uniformly random `left`, because uniform random reliably
   clumps — one draw leaves the middle bare, the next piles four into the
   right edge. Cycling the bands guarantees all three get their share, and
   the jitter within each band keeps it from looking like three columns. */
const SCATTER_BANDS = [
  { from: 2, span: 24 },   // left
  { from: 34, span: 28 },  // middle
  { from: 68, span: 24 },  // right
] as const;

function ScatterIcons({ files, count = 14 }: { files: readonly string[]; count?: number }) {
  const [pins] = useState(() => Array.from({ length: count }, (_, index) => {
    const band = SCATTER_BANDS[index % SCATTER_BANDS.length];
    return {
      file: files[index % files.length],
      /* Kept a little inside the edges so nothing is half-cut by the
         section's own bounds. */
      left: band.from + Math.random() * band.span,
      top: 1.5 + Math.random() * 93,
      size: 2.6 + Math.random() * 1.5,
    };
  }));
  return <div className="pointer-events-none absolute inset-0 z-0 overflow-hidden" aria-hidden="true">
    {pins.map((pin, index) => <PinnedIcon key={`${pin.file}-${index}`} file={pin.file} widthRem={pin.size} style={{ left: `${pin.left}%`, top: `${pin.top}%` }} />)}
  </div>;
}
